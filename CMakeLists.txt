cmake_minimum_required(VERSION 3.24)
project(edifier_n300 VERSION 0.3.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "The BLE backend requires Windows x64 and Visual Studio 2022 C++ tools.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
set(JSON_Install OFF CACHE BOOL "" FORCE)
set(N300_JSON_URL "https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz")
if(EXISTS "${CMAKE_SOURCE_DIR}/.local/dependencies/json-3.12.0.tar.xz")
    set(N300_JSON_URL "${CMAKE_SOURCE_DIR}/.local/dependencies/json-3.12.0.tar.xz")
endif()
FetchContent_Declare(nlohmann_json URL "${N300_JSON_URL}"
    URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(nlohmann_json)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_SOURCE_DIR}/resources/settings-catalog.json")
file(READ "${CMAKE_SOURCE_DIR}/resources/settings-catalog.json" N300_CATALOG)
configure_file(cmake/catalog.hpp.in generated/catalog.hpp @ONLY)
add_library(n300_engine STATIC src/native/protocol.cpp src/native/transport.cpp src/native/session.cpp)
target_include_directories(n300_engine PUBLIC src/native PRIVATE "${CMAKE_BINARY_DIR}/generated")
target_link_libraries(n300_engine PUBLIC nlohmann_json::nlohmann_json windowsapp)
target_compile_definitions(n300_engine PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(n300_engine PRIVATE /W4 /utf-8 /EHsc /bigobj)
add_library(n300 SHARED src/native/c_api.cpp)
target_include_directories(n300 PUBLIC include PRIVATE src/native)
target_link_libraries(n300 PRIVATE n300_engine)
target_compile_definitions(n300 PRIVATE N300_BUILD_DLL)
target_compile_options(n300 PRIVATE /W4 /utf-8 /EHsc)
add_executable(n300_cli src/native/cli.cpp)
target_link_libraries(n300_cli PRIVATE n300 nlohmann_json::nlohmann_json)
target_compile_options(n300_cli PRIVATE /W4 /utf-8 /EHsc)
target_compile_definitions(n300_cli PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
set_target_properties(n300_cli PROPERTIES OUTPUT_NAME n300)
set_target_properties(n300 PROPERTIES PDB_NAME n300_core)
set_target_properties(n300_cli PROPERTIES PDB_NAME n300_cli)
foreach(target n300 n300_cli)
    set_target_properties(${target} PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/artifacts/dist/$<CONFIG>"
        ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/artifacts/dist/$<CONFIG>")
endforeach()
enable_testing()
add_executable(n300_tests tests/native_tests.cpp)
target_link_libraries(n300_tests PRIVATE n300 n300_engine)
target_compile_options(n300_tests PRIVATE /W4 /utf-8 /EHsc)
set_target_properties(n300_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/artifacts/dist/$<CONFIG>")
add_test(NAME native_offline COMMAND n300_tests)
add_test(NAME cli_offline_status COMMAND n300_cli --offline status --full)
add_test(NAME cli_offline_benchmark COMMAND n300_cli --offline benchmark)
add_executable(n300_host_example examples/cpp_host.cpp)
target_link_libraries(n300_host_example PRIVATE n300)
target_compile_options(n300_host_example PRIVATE /W4 /utf-8 /EHsc)
set_target_properties(n300_host_example PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/artifacts/dist/$<CONFIG>")
add_test(NAME app_owned_worker COMMAND n300_host_example)
add_test(NAME cli_stdio COMMAND powershell -NoProfile -ExecutionPolicy Bypass -File
    "${CMAKE_SOURCE_DIR}/tests/test-stdio.ps1" -Executable "$<TARGET_FILE:n300_cli>")
install(TARGETS n300 n300_cli RUNTIME DESTINATION . ARCHIVE DESTINATION lib)
install(FILES include/n300/n300.h DESTINATION include/n300)
install(FILES "${nlohmann_json_SOURCE_DIR}/LICENSE.MIT" DESTINATION licenses RENAME nlohmann-json-LICENSE.MIT)
install(FILES cmake/sdk-readme.md DESTINATION . RENAME README.md)
install(FILES docs/api.md docs/user-guide.md DESTINATION docs)
install(FILES examples/cpp_host.cpp DESTINATION examples)
install(FILES "$<TARGET_PDB_FILE:n300>" "$<TARGET_PDB_FILE:n300_cli>" DESTINATION symbols CONFIGURATIONS Debug OPTIONAL)
