cmake_minimum_required(VERSION 3.24)
project(flesports_og98 VERSION 0.2.1 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use Windows x64 and Visual Studio 2022 C++ tools.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
set(JSON_Install OFF CACHE BOOL "" FORCE)
FetchContent_Declare(nlohmann_json
  URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz
  URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(nlohmann_json)
add_library(og98_engine STATIC src/protocol.cpp src/transport.cpp src/session.cpp src/keymap.cpp)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/layout.json")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/assets/layout.json" OG98_LAYOUT_JSON)
configure_file(cmake/og98_layout.hpp.in generated/og98_layout_generated.hpp @ONLY)
target_include_directories(og98_engine PUBLIC src "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(og98_engine PUBLIC nlohmann_json::nlohmann_json hid setupapi)
target_compile_definitions(og98_engine PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
add_library(og98 SHARED src/c_api.cpp)
target_include_directories(og98 PUBLIC include)
target_link_libraries(og98 PRIVATE og98_engine)
target_compile_definitions(og98 PRIVATE OG98_BUILD_DLL)
add_executable(og98_cli src/cli.cpp)
target_link_libraries(og98_cli PRIVATE og98 nlohmann_json::nlohmann_json)
target_compile_definitions(og98_cli PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
set_target_properties(og98_cli PROPERTIES OUTPUT_NAME og98)
set_target_properties(og98 PROPERTIES PDB_NAME og98_core)
set_target_properties(og98_cli PROPERTIES PDB_NAME og98_cli)
enable_testing()
add_executable(og98_tests tests/core_tests.cpp)
target_link_libraries(og98_tests PRIVATE og98_engine og98)
add_test(NAME protocol_and_api COMMAND og98_tests)
add_test(NAME cli_offline_status COMMAND og98_cli --offline --no-log status)
add_test(NAME cli_offline_write COMMAND og98_cli --offline --no-log brightness 2)
add_test(NAME cli_stdio COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass -File
  "${CMAKE_SOURCE_DIR}/tests/stdio.ps1" -Executable "$<TARGET_FILE:og98_cli>")
foreach(target og98_engine og98 og98_cli og98_tests)
  target_compile_options(${target} PRIVATE /W4 /utf-8 /EHsc)
  set_target_properties(${target} PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/artifacts/dist/$<CONFIG>"
    ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/artifacts/dist/$<CONFIG>")
endforeach()
install(TARGETS og98 og98_cli RUNTIME DESTINATION . ARCHIVE DESTINATION lib)
install(FILES include/og98/og98.h DESTINATION include/og98)
install(FILES cmake/package-readme.md DESTINATION . RENAME README.md)
install(DIRECTORY docs/ DESTINATION docs)
install(FILES research/official-manifest.json DESTINATION research)
install(FILES "${nlohmann_json_SOURCE_DIR}/LICENSE.MIT" DESTINATION licenses RENAME nlohmann-json-LICENSE.MIT)
install(FILES "$<TARGET_PDB_FILE:og98>" "$<TARGET_PDB_FILE:og98_cli>" DESTINATION symbols CONFIGURATIONS Debug OPTIONAL)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../sdk/module.cmake")
  include(../sdk/module.cmake)
  deskhub_module(deskhub-og98 "${CMAKE_CURRENT_SOURCE_DIR}/adapter.cpp" og98)
  add_executable(og98_reconnect_tests tests/reconnect_tests.cpp adapter.cpp)
  target_include_directories(og98_reconnect_tests PRIVATE include ../sdk/include)
  target_link_libraries(og98_reconnect_tests PRIVATE nlohmann_json::nlohmann_json)
  target_compile_definitions(og98_reconnect_tests PRIVATE OG98_BUILD_DLL DESKHUB_MODULE_ID="flesports-og98" DESKHUB_MODULE_VERSION="${PROJECT_VERSION}")
  target_compile_options(og98_reconnect_tests PRIVATE /W4 /utf-8 /EHsc)
  add_test(NAME reconnect_lifecycle COMMAND og98_reconnect_tests)
endif()
