cmake_minimum_required(VERSION 3.24)
project(logitech_ghub VERSION 0.2.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use Windows x64 and Visual Studio 2022 C++ tools.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
set(JSON_Install OFF CACHE BOOL "" FORCE)
FetchContent_Declare(nlohmann_json
  URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz
  URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(nlohmann_json)
add_library(ghub_engine STATIC src/client.cpp src/settings.cpp src/simulator.cpp src/macros.cpp)
target_include_directories(ghub_engine PUBLIC src ../sdk/include)
target_link_libraries(ghub_engine PUBLIC nlohmann_json::nlohmann_json winhttp)
target_compile_definitions(ghub_engine PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(ghub_engine PRIVATE /W4 /utf-8 /EHsc)
include(../sdk/module.cmake)
deskhub_module(deskhub-ghub "${CMAKE_CURRENT_SOURCE_DIR}/adapter.cpp" ghub_engine)
enable_testing()
add_executable(ghub_tests tests/settings.cpp)
target_link_libraries(ghub_tests PRIVATE ghub_engine)
target_compile_options(ghub_tests PRIVATE /W4 /utf-8 /EHsc)
add_test(NAME ghub_settings COMMAND ghub_tests)
add_test(NAME ghub_transport COMMAND ${CMAKE_COMMAND} -E env "DESKHUB_GHUB_TEST_EXE=$<TARGET_FILE:deskhub-ghub>"
  node --test "${CMAKE_CURRENT_SOURCE_DIR}/tests/wire.mjs")
install(FILES "${nlohmann_json_SOURCE_DIR}/LICENSE.MIT" DESTINATION . RENAME nlohmann-json-LICENSE.MIT)
